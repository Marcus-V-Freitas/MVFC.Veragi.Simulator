paths:
  /webhooks/card-receivable/schedules:
    post:
      tags:
        - "Card Receivable - Schedule Webhooks"
      summary: "Receive card receivable schedule webhook notification"
      description: |
        Veragi sends **asynchronous schedule query results** (card receivable schedules)
        to this webhook endpoint after a schedule query request has been processed.

        The partner must expose this endpoint and handle the incoming data.

        **Idempotency:**
        - The `Idempotency-Key` header uniquely identifies the webhook event.
        - Retries for the same event will reuse the same key.
        - The partner SHOULD treat repeated calls with the same `Idempotency-Key`
          as idempotent and avoid duplicating side effects.

        **Typical usage:**
        - Persist the schedule data received in `scheduleQueryData`.
        - Use the `requestId` to correlate with the original request on the partner side (if applicable).
        - Use the `status` field to determine whether the schedule was successfully obtained
          (`PROCESSED`), is still being processed (`PROCESSING`), or could not be obtained (`ERROR`).
      operationId: "crScheduleWebhookReceive"
      parameters:
        - name: "Idempotency-Key"
          in: "header"
          required: true
          description: |
            Unique identifier for the webhook event.

            - Must be treated as an **idempotency key**.
            - If the same key is received more than once, the partner SHOULD ensure
              that the event processing is not duplicated.
          schema:
            type: "string"
            format: "uuid"
          example: "4b8e2c4a-0c43-4a81-9d0c-5e1348b8bb2c"
      requestBody:
        required: true
        description: "Schedule webhook payload with the schedule query status and schedule data."
        content:
          application/json:
            schema:
              $ref: "#/components/schemas/ScheduleWebhookNotification"
            examples:
              ScheduleWebhookProcessed:
                summary: "Schedule processed successfully"
                description: "Example of a webhook when the schedule query has been fully processed."
                value:

                  detail: "The schedule query request has been successfully processed."

                    originType: "ONLINE"

                    merchantCnpj: "23399607000191"
                    acquirers:

                        paymentArrangements:
                          - code: "ACC"
                            receivableUnits:
                              - settlementDate: "2025-11-01"
                                totalAmount: 7303.62
                                freeAmount: 0.0
                              - settlementDate: "2025-11-02"
                                totalAmount: 6342.42
                                freeAmount: 0.0
                          - code: "DCC"
                            receivableUnits:
                              - settlementDate: "2025-11-01"
                                totalAmount: 3033.35
                                freeAmount: 3033.35
                          - code: "VCC"
                            receivableUnits: []
              ScheduleWebhookError:
                summary: "Schedule retrieval error"
                description: "Example of a webhook when the schedule could not be obtained."
                value:
                  status: "ERROR"
                  detail: "It was not possible to retrieve the schedule from the receivables registry."
                  scheduleQueryData:
                    requestId: "58bd86ff-6b68-41c8-a318-37873fcbf609"
                    originType: "ONLINE"
                    updatedAt: "2025-11-18T08:01:46Z"
                    merchantCnpj: "23399607000191"
                    acquirers: null
      responses:
        "200":
          description: >
            Webhook successfully received and accepted for processing.

            The partner does not need to return a body; an empty 200 OK response is sufficient.

    # ---------------------------------------
    # Webhook payload schemas
    # ---------------------------------------
    ScheduleWebhookNotification:
      description: >
        Payload sent by Veragi to notify the partner about the status and result
        of a schedule query request.
      type: "object"
      required:
        - status
        - detail
        - scheduleQueryData
      properties:
        status:
          $ref: "#/components/schemas/ScheduleQueryStatusType"
        detail:
          type: "string"
          description: "Detailed message describing the current status of the schedule query request."
          example: "The schedule query request has been successfully processed."
        scheduleQueryData:
          $ref: "#/components/schemas/ScheduleData"

    ScheduleData:
      description: "Schedule data returned by the receivables registry for a given query request."
      type: "object"
      required:
        - requestId
        - originType
        - updatedAt
        - merchantCnpj
      properties:
        requestId:
          type: "string"
          format: "uuid"
          description: "Unique identifier of the schedule query request."
          example: "31871a08-5d99-4e01-877f-7efb2804e84d"
        originType:
          $ref: "#/components/schemas/ScheduleQueryOriginType"
        updatedAt:
          type: "string"
          format: "date-time"
          description: "UTC timestamp when the schedule data was last updated."
          example: "2025-11-04T13:21:37Z"
        merchantCnpj:
          type: "string"
          description: "Merchant CNPJ (14 digits or 12 alphanumeric + 2 numeric check digits). No punctuation."
          example: "23399607000191"
        acquirers:
          type: "array"
          description: "List of acquirers/sub-acquirers and their associated payment arrangements."
          items:
            $ref: "#/components/schemas/SchedulePaymentAcquirer"

    ScheduleQueryOriginType:
      description: >
        Indicates whether the schedule data was obtained through a direct query (ONLINE)
        or from the latest available batch processing (BATCH).
      type: "string"
      enum:
        - "ONLINE"
        - "BATCH"
      example: "ONLINE"

    ScheduleQueryStatusType:
      description: "Indicates the current processing status of the schedule query request."
      type: "string"
      enum:
        - "PROCESSED"
        - "PROCESSING"
        - "ERROR"
      example: "PROCESSED"

    SchedulePaymentAcquirer:
      description: "Acquirer or sub-acquirer and its associated payment arrangements for the schedule."
      type: "object"
      required:
        - cnpj
      properties:
        cnpj:
          type: "string"
          description: "Acquirer/sub-acquirer CNPJ (14 digits or 12 alphanumeric + 2 numeric). No punctuation."
          example: "82413081000116"
        paymentArrangements:
          type: "array"
          description: "List of payment arrangements (card brands/modes) associated with this acquirer."
          items:
            $ref: "#/components/schemas/SchedulePaymentArrangement"

    SchedulePaymentArrangement:
      description: "Payment arrangement and its receivable units grouped by settlement date."
      type: "object"
      required:
        - code
      properties:
        code:
          type: "string"
          pattern: "^(?:[A-Z]{3}|999)$"
          description: "Payment arrangement code in `XXX` format (three uppercase letters) or the wildcard `999`."
          example: "VCC"
        receivableUnits:
          type: "array"
          description: "Receivable units (amounts) associated with this payment arrangement per settlement date."
          items:
            $ref: "#/components/schemas/ScheduleReceivableUnit"

    ScheduleReceivableUnit:
      description: "Receivable amounts for a specific settlement date."
      type: "object"
      required:
        - settlementDate
        - totalAmount
        - freeAmount
      properties:
        settlementDate:
          type: "string"
          format: "date"
          description: "Settlement date in ISO 8601 format (yyyy-MM-dd)."
          example: "2025-11-01"
        totalAmount:
          type: "number"
          minimum: 0
          description: "Total BRL amount of receivables scheduled for this date."
          example: 7303.62
        freeAmount:
          type: "number"
          minimum: 0
          description: "Free (unencumbered) BRL amount of receivables for this date."
          example: 0.0
